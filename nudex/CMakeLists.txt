cmake_minimum_required(VERSION 3.16)
project(nudex_order_test CXX)
# Upstream NuDEX builds against ROOT (TRandom2) by default: include/NuDEXRandom.hh sets COMPILATIONTYPE 1.
find_package(ROOT REQUIRED COMPONENTS MathCore)
find_package(Git REQUIRED)

set(NUDEX_URL https://github.com/UIN-CIEMAT/NuDEX.git CACHE STRING "NuDEX repository")
set(NUDEX_REV 71d5d4d8 CACHE STRING "NuDEX revision under test")
set(src ${CMAKE_BINARY_DIR}/nudex-upstream)
set(fixed ${CMAKE_BINARY_DIR}/nudex-fixed)
if(NOT EXISTS ${src}/src/NuDEXStatisticalNucleus.cc)
  execute_process(COMMAND ${GIT_EXECUTABLE} clone -q ${NUDEX_URL} ${src} COMMAND_ERROR_IS_FATAL ANY)
  execute_process(COMMAND ${GIT_EXECUTABLE} -C ${src} checkout -q ${NUDEX_REV} COMMAND_ERROR_IS_FATAL ANY)
endif()
if(NOT EXISTS ${fixed}/src/NuDEXStatisticalNucleus.cc)
  file(COPY ${src}/include ${src}/src DESTINATION ${fixed})
  # git init first: inside another checkout, git apply skips paths outside the current directory.
  execute_process(COMMAND ${GIT_EXECUTABLE} init -q WORKING_DIRECTORY ${fixed} COMMAND_ERROR_IS_FATAL ANY)
  execute_process(COMMAND ${GIT_EXECUTABLE} apply ${CMAKE_CURRENT_SOURCE_DIR}/../patches/nudex-reseed-icc.patch
                  WORKING_DIRECTORY ${fixed} COMMAND_ERROR_IS_FATAL ANY)
endif()
file(STRINGS ${fixed}/include/NuDEXStatisticalNucleus.hh applied REGEX "theICC->SetRandom4Seed\\(seed\\)")
if(NOT applied)
  message(FATAL_ERROR "patches/nudex-reseed-icc.patch did not apply to ${fixed}")
endif()
execute_process(COMMAND ${GIT_EXECUTABLE} -C ${src} rev-parse --short=7 HEAD OUTPUT_VARIABLE rev OUTPUT_STRIP_TRAILING_WHITESPACE)
message(STATUS "NuDEX revision: ${rev}")

foreach(variant upstream fixed)
  file(GLOB lib_src ${CMAKE_BINARY_DIR}/nudex-${variant}/src/*.cc)
  add_executable(order_test_${variant} order_test.cc ${lib_src})
  target_include_directories(order_test_${variant} PRIVATE ${CMAKE_BINARY_DIR}/nudex-${variant}/include)
  target_link_libraries(order_test_${variant} PRIVATE ROOT::MathCore)
  target_compile_features(order_test_${variant} PRIVATE cxx_std_17)
endforeach()
